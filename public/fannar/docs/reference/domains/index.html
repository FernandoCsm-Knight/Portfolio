    Domain modules¶

    One entry point per kind of data, each a subclass of Interpreter that only sets defaults and handles the inputs. The modules load on first access (fa.tabular, fa.image, fa.text, fa.tree), so import fannar stays light.

    def as_table(X, y=None, feature_names: Sequence[str] | None = None, class_names: Sequence[str] | None = None,
             classes: Sequence | None = None, dtype=np.float32):
    """``(X, y, feature_names, class_names)`` from arrays or pandas objects.

    Explicit ``feature_names``/``class_names`` take precedence over what the DataFrame and the labels
    provide. ``X`` becomes a ``dtype`` array (``float32`` by default, the usual dtype of a model).
    """
    if hasattr(X, "columns") and feature_names is None:
        feature_names = [str(c) for c in X.columns]
    if hasattr(X, "to_numpy"):
        X = X.to_numpy()
    if not hasattr(X, "detach"):  # leave tensors to the model's own dtype handling
        X = np.asarray(X, dtype=dtype)
    y, names = encode_labels(y, classes)
    return X, y, feature_names, class_names if class_names is not None else names

    def encode_labels(y, classes: Sequence | None = None) -> tuple[np.ndarray | None, list[str] | None]:
    """``y`` as class indices ``0 .. K-1`` plus the class names, when ``y`` is not already numeric.

    A pandas categorical keeps its category order. Strings, objects and booleans are matched against
    ``classes`` when given (e.g. ``tree.classes_``), otherwise against their sorted unique values, the
    same order as scikit-learn's ``LabelEncoder``. Numeric labels are returned unchanged with no names.
    """
    if y is None:
        return None, None
    cat = getattr(y, "cat", None)
    if cat is not None and str(getattr(y, "dtype", "")) == "category":
        return np.asarray(cat.codes, dtype=np.int64), [str(c) for c in cat.categories]
    y = np.asarray(y)
    if classes is not None:
        classes = np.asarray(classes)
        if np.issubdtype(classes.dtype, np.integer) and np.array_equal(classes, np.arange(len(classes))):
            return y, None
        index = {c: t for t, c in enumerate(classes.tolist())}
        unknown = sorted({str(c) for c in y.tolist() if c not in index})
        if unknown:
            raise ValueError(f"labels {unknown[:5]} are not among the model's classes {classes.tolist()}")
        return np.array([index[c] for c in y.tolist()], dtype=np.int64), [str(c) for c in classes]
    if np.issubdtype(y.dtype, np.number):
        return y, None
    names, codes = np.unique(y, return_inverse=True)
    return codes.astype(np.int64), [str(c) for c in names]

    def block_names(model) -> list[str]:
    """Names of the children of the longest ``nn.ModuleList`` in ``model``: the transformer blocks.

    Raises ``ValueError`` when the model has no ``nn.ModuleList``; pass ``layers=`` explicitly then.
    """
    _need_torch()
    lists = [(name, m) for name, m in model.named_modules() if isinstance(m, nn.ModuleList) and len(m)]
    if not lists:
        raise ValueError("no nn.ModuleList found in the model; pass `layers=` explicitly")
    name, blocks = max(lists, key=lambda item: len(item[1]))  # first of the longest, in module order
    return [f"{name}.{t}" if name else str(t) for t in range(len(blocks))]

    def __init__(self, tree, X, y=None, *, X_train=None, scale=None, depths: Sequence[int] | None = None,
             feature_names: Sequence[str] | None = None, class_names: Sequence[str] | None = None,
             **kwargs) -> None:
    classes = getattr(tree, "classes_", None)
    X, y, feature_names, class_names = as_table(X, y, feature_names, class_names, classes=classes,
                                                dtype=np.float64)
    if class_names is None and classes is not None and y is not None and not np.array_equal(
            np.asarray(classes), np.arange(len(classes))):
        class_names = [str(c) for c in classes]
    if scale is None and X_train is not None:
        Xt = np.asarray(X_train.to_numpy() if hasattr(X_train, "to_numpy") else X_train, dtype=np.float64)
        scale = np.where(Xt.std(0) > 0, Xt.std(0), 1.0)
    reference = kwargs.pop("reference_model", None)
    extractor = _IndexedTreeSources(tree, scale=scale, depths=depths)
    super().__init__(tree, X, y, extractor=extractor, feature_names=feature_names,
                     class_names=class_names, **kwargs)
    if reference is not None:  # another tree (e.g. a shallower or randomised one) as the floor
        self.reference_extractor = _IndexedTreeSources(reference, scale=scale, depths=depths)
